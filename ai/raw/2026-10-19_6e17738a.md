def _stab(tracks, key="fam", min_len=5, pred=None):
    pair_eq=pair_n=strict=n=0
    lens=[]
    for tr in tracks:
        if pred is not None and not pred(tr): continue
        labs=[d.get(key) for d in tr]
        if any(x is None for x in labs) or len(labs)<min_len: continue
        n+=1; lens.append(len(labs))
        eq=[1 if labs[i]==labs[i+1] else 0 for i in range(len(labs)-1)]
        pair_eq+=sum(eq); pair_n+=len(eq)
        if all(eq): strict+=1
    return {...}

def _stab_shuf(tracks, key, min_len, pred, reps, rs):
    frames={}
    for tr in tracks:
        for d in tr: frames.setdefault(d["fi"],[]).append(d)
    keys=sorted(frames)
    vals=[[d.get(key) for d in frames[f]] for f in keys]
    acc=[]
    for _ in range(reps):
        for f,v in zip(keys,vals):
            p=rs.permutation(len(v))
            for pos,d in zip(p,frames[f]): d["_shuf"]=v[pos]
        s=_stab(tracks,key="_shuf",min_len=min_len,pred=pred)
        if s["consec"] is not None: acc.append(s["consec"])
    for tr in tracks:
        for d in tr: d.pop("_shuf",None)
    return float(np.mean(acc)) if acc else None
